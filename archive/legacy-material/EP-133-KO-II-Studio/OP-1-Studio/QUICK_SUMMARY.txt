In any React component:

import { useProfileStore, useDeviceStore, useLibraryStore } from '@/app/core/store';

export function MyComponent() {
  // Read state (reactive)
  const profile = useProfileStore(s => s.profile);
  const isConnected = useDeviceStore(s => s.isConnected);
  const libraryPath = useLibraryStore(s => s.libraryPath);

  // Update state
  const setProfile = useProfileStore(s => s.setProfile);
  const connect = useDeviceStore(s => s.connect);
  const setLibraryPath = useLibraryStore(s => s.setLibraryPath);

  return (
    <div>
      <p>Profile: {profile.name}</p>
      <p>Connected: {isConnected ? 'Yes' : 'No'}</p>
      <p>Library: {libraryPath || 'Not set'}</p>
      
      <button onClick={() => setProfile({ masterVolume: 80 })}>
        Set Volume to 80
      </button>
      <button onClick={() => connect({ name: 'OP-1', version: '246' })}>
        Connect Device
      </button>
      <button onClick={() => setLibraryPath('/home/user/OP-1-Library')}>
        Set Library Path
      </button>
    </div>
  );
}
